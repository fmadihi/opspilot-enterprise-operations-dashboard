function App() {

  const location = useLocation();
  const pageTitle = nav.find(n => n.path === location.pathname)?.label || "Overview";
 
  const updateTasks = (next: Task[]) => { setTasks(next); localStorage.setItem("opspilot-tasks", JSON.stringify(next)); };

          <Route path="/projects" element={<Projects projects={projects}/>}/>
